definition{
  name: decideGrant,
  parameters: [organCode : Nat, capCode : Nat, offlineMode : Nat, explicitDenied : Nat],
  result: Nat,
  body: (if (explicitDenied =? 1) then 90 else (if (8 <=? organCode) then 92 else (if (5 <=? capCode) then 93 else (if ((capCode =? 2) && (offlineMode =? 1)) then 91 else 1))))
}

definition{
  name: breaks_denial_dominates_grant,
  parameters: [organCode : Nat, capCode : Nat, offlineMode : Nat],
  result: Bool,
  body: (not ((decideGrant organCode capCode offlineMode 1) =? 90))
}

theorem{
  name: denial_dominates_grant,
  parameters: [organCode : Nat, capCode : Nat, offlineMode : Nat],
  statement: ((breaks_denial_dominates_grant organCode capCode offlineMode) = false),
  proof: linear_arithmetic{
    definitions: [breaks_denial_dominates_grant, decideGrant]
  },
  axioms: [Quot.sound, propext]
}

definition{
  name: breaks_offline_denies_network,
  parameters: [organCode : Nat],
  result: Bool,
  body: ((organCode <=? 7) && (not ((decideGrant organCode 2 1 0) =? 91)))
}

theorem{
  name: offline_denies_network,
  parameters: [organCode : Nat],
  statement: ((breaks_offline_denies_network organCode) = false),
  proof: linear_arithmetic{
    definitions: [breaks_offline_denies_network, decideGrant]
  },
  axioms: [Quot.sound, propext]
}

definition{
  name: breaks_stateful_organ_admits_storage,
  parameters: [organCode : Nat, offlineMode : Nat],
  result: Bool,
  body: (((3 <=? organCode) && (organCode <=? 6)) && (not ((decideGrant organCode 1 offlineMode 0) =? 1)))
}

theorem{
  name: stateful_organ_admits_storage,

definition{
  name: formatGrantCode,
  parameters: [code : Nat, capTok : Bytes],
  result: Bytes,
  body: (if (code =? 1) then (prim append 0x616c6c6f7709 capTok : Bytes) else
    (if (code =? 90) then 0x726566757365643a2064656e69656420627920464b2d3037 else
    (if (code =? 91) then 0x726566757365643a206f66666c696e65206d6f64652064656e696573206e6574776f726b else

definition{
  name: handleGrant,
  parameters: [args : Bytes],
  result: Bytes,
  body: (let oTok : Bytes := (VerdantUI.Protocol::field0 args);
    (let r1 : Bytes := (VerdantUI.Protocol::afterField0 args);
    (let cTok : Bytes := (VerdantUI.Protocol::field0 r1);
    (let r2 : Bytes := (VerdantUI.Protocol::afterField0 r1);
    (let offTok : Bytes := (VerdantUI.Protocol::field0 r2);
    (let denTok : Bytes := (VerdantUI.Protocol::afterField0 r2);
    (formatGrantCode (decideGrant (encodeOrgan oTok) (encodeCapability cTok) (VerdantUI.Protocol::parseBoolBit offTok) (VerdantUI.Protocol::parseBoolBit denTok)) cTok)))))))
}
