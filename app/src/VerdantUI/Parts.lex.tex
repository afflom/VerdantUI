\title{Text application model}
\begin{semanticmodule}
definition{
  name: decidePart,
  parameters: [partCode : Nat, verbCode : Nat],
  result: Nat,
  body: (if (8 <=? partCode) then 90 else (if (5 <=? verbCode) then 91 else 1))
}

definition{
  name: breaks_all_registered_parts_answer_status,
  parameters: [partCode : Nat],
  result: Bool,
  body: ((partCode <=? 7) && (not ((decidePart partCode 0) =? 1)))
}

theorem{
  name: all_registered_parts_answer_status,
  parameters: [partCode : Nat],
  statement: ((breaks_all_registered_parts_answer_status partCode) = false),
  proof: linear_arithmetic{
    definitions: [breaks_all_registered_parts_answer_status, decidePart]
  },
  axioms: [Quot.sound, propext]
}

definition{
  name: breaks_unknown_part_refused,
  parameters: [partCode : Nat, verbCode : Nat],
  result: Bool,
  body: ((8 <=? partCode) && (not ((decidePart partCode verbCode) =? 90)))
}

theorem{
  name: unknown_part_refused,
  parameters: [partCode : Nat, verbCode : Nat],
  statement: ((breaks_unknown_part_refused partCode verbCode) = false),
  proof: linear_arithmetic{
    definitions: [breaks_unknown_part_refused, decidePart]
  },
  axioms: [Quot.sound, propext]
}

definition{
  name: breaks_unknown_verb_refused,
  parameters: [partCode : Nat, verbCode : Nat],
  result: Bool,
  body: (((partCode <=? 7) && (5 <=? verbCode)) && (not ((decidePart partCode verbCode) =? 91)))
}

theorem{
  name: unknown_verb_refused,
  parameters: [partCode : Nat, verbCode : Nat],
  statement: ((breaks_unknown_verb_refused partCode verbCode) = false),
  proof: linear_arithmetic{
    definitions: [breaks_unknown_verb_refused, decidePart]
  },
  axioms: [Quot.sound, propext]
}

definition{
  name: encodeVerb,
  parameters: [tok : Bytes],
  result: Nat,
  body: (if (VerdantUI.Protocol::bytesEq tok 0x737461747573) then 0 else
    (if (VerdantUI.Protocol::bytesEq tok 0x696e707574) then 1 else
    (if (VerdantUI.Protocol::bytesEq tok 0x726566696e65) then 2 else
    (if (VerdantUI.Protocol::bytesEq tok 0x7075626c697368) then 3 else
    (if (VerdantUI.Protocol::bytesEq tok 0x64656c657465) then 4 else 5)))))
}

definition{
  name: handlePart,
  parameters: [args : Bytes],
  result: Bytes,
  body: (let pTok : Bytes := (VerdantUI.Protocol::field0 args);
    (let vTok : Bytes := (VerdantUI.Protocol::afterField0 args);
    (let code : Nat := (decidePart (VerdantUI.Grants::encodeOrgan pTok) (encodeVerb vTok));
    (if (code =? 1) then (prim append 0x6f6b09 (prim append pTok (prim append 0x09 vTok : Bytes) : Bytes) : Bytes) else
      (if (code =? 90) then 0x726566757365643a20756e6b6e6f776e2070617274 else
      0x726566757365643a20756e6b6e6f776e20706172742076657262)))))
}
